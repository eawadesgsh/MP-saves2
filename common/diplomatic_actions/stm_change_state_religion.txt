
change_state_religion = {
	groups= {
		overlord
	}
	
	requires_approval = no
	should_notify_third_parties = no
	
	texture = "gfx/interface/icons/diplomatic_action_icons/da_appoint_colonial_governor.dds"
	confirmation_sound = "event:/SFX/UI/MapInteraction/SubjectInteractions/diplomatic_action_subjects_04_appoint_governor_benign"
	
	unlocking_technologies = {
		dialectics
	}
	
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			OR = {
				is_subject_type = subject_type_colony
				is_subject_type = subject_type_puppet
			}
		}
		scope:target_country = {
			NOT = { religion = root.religion }
		}
		is_player = yes
	}
	
	possible = {
		is_revolutionary = no
		in_default = no
		NOT = { has_war_with = scope:target_country }
		OR = { 
			NOT = { has_law = law_type:law_total_separation }
			country_has_state_religion = rel:atheist
		}
		scope:target_country = {
		    NOT = { has_law = law_type:law_total_separation }
			NOT = { 
				AND = {
					has_law = law_type:law_state_atheism
					country_has_state_religion = rel:atheist	# This extra check is because mods can mess things up, and it stops a messed up situation getting locked in.
				}
			}
		}
		scope:target_country.relations:root > relations_threshold:cordial
		has_diplomatic_relevance = scope:target_country
	}
	
	accept_effect = {
		scope:target_country = {
			set_state_religion = root.religion
			add_liberty_desire = 5
		}
	}
	
	
}
